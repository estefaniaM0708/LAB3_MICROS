#include <Wire.h>

int MPU = 0x68; // dirección del sensor en el bus I2C

// variables para guardar lo que se lee del sensor (
int16_t axRaw, ayRaw, azRaw;
int16_t gxRaw, gyRaw, gzRaw;
int16_t tempRaw;

// timer para que se active cada 50 ms (20 Hz)
hw_timer_t *miTimer = NULL;
volatile bool listo = false; //  "ya toca leer otra vez"

void IRAM_ATTR cadaCiclo() {
  listo = true; // solo avisamos, no leemos aquí
}

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22); // SDA = 21, SCL = 22

  // despertar
  Wire.beginTransmission(MPU);
  Wire.write(0x6B); // registro PWR_MGMT_1
  Wire.write(0);
  Wire.endTransmission();

  // timer a 1,000,000 Hz (1 tick = 1 microsegundo)
  miTimer = timerBegin(1000000);
  timerAttachInterrupt(miTimer, &cadaCiclo);
  timerAlarm(miTimer, 50000, true, 0); // 50000 us = 50 ms = 20 Hz
}

void loop() {
  if (listo) {
    listo = false;
    leerSensor();
    enviarDatos();
  }
}

void leerSensor() {
  Wire.beginTransmission(MPU);
  Wire.write(0x3B); // primer registro de datos (ACCEL_XOUT_H)
  Wire.endTransmission(false);
  Wire.requestFrom(MPU, 14, true); // pedimos los 14 bytes de golpe

  axRaw = Wire.read() << 8 | Wire.read();
  ayRaw = Wire.read() << 8 | Wire.read();
  azRaw = Wire.read() << 8 | Wire.read();
  tempRaw = Wire.read() << 8 | Wire.read();
  gxRaw = Wire.read() << 8 | Wire.read();
  gyRaw = Wire.read() << 8 | Wire.read();
  gzRaw = Wire.read() << 8 | Wire.read();
}

void enviarDatos() {
  // convertir los valores crudos a unidades reales (según el datasheet)
  float ax = axRaw / 16384.0; // g
  float ay = ayRaw / 16384.0;
  float az = azRaw / 16384.0;
  float gx = gxRaw / 131.0;   // grados/segundo
  float gy = gyRaw / 131.0;
  float gz = gzRaw / 131.0;
  float temp = tempRaw / 340.0 + 36.53; // grados Celsius

  // enviar todo en una sola línea separada por comas
  Serial.print(ax); Serial.print(",");
  Serial.print(ay); Serial.print(",");
  Serial.print(az); Serial.print(",");
  Serial.print(gx); Serial.print(",");
  Serial.print(gy); Serial.print(",");
  Serial.print(gz); Serial.print(",");
  Serial.println(temp);
}
